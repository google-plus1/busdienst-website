      <button onclick="login()">Inloggen</button>
    </div>

    <div id="main-section" style="display:none;">
      <p id="welcome"></p>
      <button id="clockInBtn" onclick="clockIn()">🟢 Inklokken</button>
      <button id="clockOutBtn" class="logout" onclick="clockOut()">🔴 Uitklokken</button>
      <div class="status" id="statusText">Status: Uit dienst</div>
      <button onclick="logout()">Afmelden</button>
    </div>

  <script>
    let loggedInUser = null;
    let inDienst = false;

    function login() {
      const username = document.getElementById('username').value.trim();
      if (username === "") {
        alert("Vul een gebruikersnaam in!");
        return;
      }
      loggedInUser = username;
      document.getElementById('login-section').style.display = 'none';
      document.getElementById('main-section').style.display = 'block';
      document.getElementById('welcome').innerText = "Ingelogd als: " + username;
      updateStatus();
    }

    function clockIn() {
      if (!inDienst) {
        inDienst = true;
        updateStatus();
      }
    }

    function clockOut() {
      if (inDienst) {
        inDienst = false;
        updateStatus();
      }
    }

    function updateStatus() {
      const statusText = document.getElementById('statusText');
      statusText.innerText = inDienst ? "Status: 🟢 In dienst" : "Status: 🔴 Uit dienst";
    }

    function logout() {
      loggedInUser = null;
      inDienst = false;
      document.getElementById('login-section').style.display = 'block';
      document.getElementById('main-section').style.display = 'none';
      document.getElementById('username').value = "";
    }
  </script>
